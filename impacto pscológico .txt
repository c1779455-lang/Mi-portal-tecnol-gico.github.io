      <h2 class="titulo-destacado">El impacto emocional de la hiperconectividad</h2>
      <p>
        Durante la adolescencia, el cerebro experimenta cambios profundos en las áreas encargadas de las emociones y la recompensa social. El uso desmedido de redes sociales y dispositivos digitales puede alterar este desarrollo natural, generando vulnerabilidad ante la presión de grupo virtual, la comparación constante con vidas idealizadas y la necesidad de validación externa a través de "likes".
      </p>

    <section id="efectos-emocionales" class="tarjeta">
      <h3 class="titulo-destacado">Principales trastornos y riesgos psicológicos</h3>
      <p>
        Entre los fenómenos psicológicos más estudiados en jóvenes destacan el <strong>FOMO</strong> (Fear of Missing Out o miedo a perderse de algo), que provoca ansiedad constante por estar conectado; la <strong>baja autoestima</strong> derivada de los filtros estéticos y la perfección fingida en internet; y el aislamiento social debido a la sustitución de la convivencia cara a cara por interacciones virtuales.
      </p>
      <p>
        Asimismo, el ciberacoso representa un factor crítico que daña severamente la estabilidad emocional, pudiendo desencadenar cuadros de depresión severa si no se detecta y atiende de manera oportuna con apoyo psicológico y familiar.
      </p>
    </section>

    const tiposFlores = ["🌸", "🌼", "🌺"];

    for (let i = 0; i < cantidadFlores; i++) {
      const flor = document.createElement("div");
      flor.classList.add("flor-rotando");
      flor.innerText = tiposFlores[i % tiposFlores.length];
      
      flor.style.top = Math.random() * 100 + "vh";
      flor.style.left = Math.random() * 100 + "vw";
      
      const tamano = Math.floor(Math.random() * 25) + 20;
      flor.style.fontSize = tamano + "px";
      
      const duracion = Math.floor(Math.random() * 12) + 8;
      flor.style.animationDuration = duracion + "s";

      contenedorFlores.appendChild(flor);
    }

    // Animación de bienvenida y aparición de tarjetas
    const checkboxBienvenida = document.getElementById("cerrar-bienvenida");
    const textoTitulo = "Impacto Psicológico Digital";
    const elementoTitulo = document.getElementById("titulo-animado");
    let animacionIniciada = false;

    function iniciarAnimaciones() {
      if (animacionIniciada) return;
      animacionIniciada = true;

      let indice = 0;
      function escribirTitulo() {
        if (indice < textoTitulo.length) {
          elementoTitulo.innerHTML += textoTitulo.charAt(indice);
          indice++;
          setTimeout(escribirTitulo, 100);
        }
      }
      escribirTitulo();

      const tarjetas = document.querySelectorAll('.tarjeta');
      tarjetas.forEach((tarjeta, index) => {
        setTimeout(() => {
          tarjeta.classList.add('animada');
        }, 300 * (index + 1));
      });
    }

    checkboxBienvenida.addEventListener("change", function() {
      if (checkboxBienvenida.checked) {
        iniciarAnimaciones();
      }
    });

    // Control del menú desplegable
    const botonMenu = document.querySelector(".dropbtn");
    const listaMenu = document.querySelector(".contenido-dropdown");

    botonMenu.addEventListener("click", function(evento) {
      evento.stopPropagation();
      listaMenu.classList.toggle("mostrar");
    });

    window.addEventListener("click", function() {
      if (listaMenu.classList.contains("mostrar")) {
        listaMenu.classList.remove("mostrar");
      }
    });
  </script>
