  <script>
    if ('scrollRestoration' in history) {
      history.scrollRestoration = 'manual';
    }

    function irAlInicio() {
      window.scrollTo(0, 0);
    }

    function reiniciarCuestionario() {
      const formulario = document.getElementById("form-encuesta");
      if (formulario) {
        formulario.reset();
      }
      document.querySelectorAll('.pregunta-encuesta').forEach(bloque => {
        bloque.classList.remove('pregunta-incompleta');
      });
      cerrarModalResultado();
    }

    window.addEventListener("pageshow", function() {
      irAlInicio();
      reiniciarCuestionario();
    });

    window.addEventListener("load", irAlInicio);

    // Flores rotando por todo el fondo
    const contenedorFlores = document.getElementById("contenedor-flores");
    const cantidadFlores = 25;
    const tiposFlores = ["🌸", "🌼", "🌺", "💠"];

    for (let i = 0; i < cantidadFlores; i++) {
      const flor = document.createElement("div");
      flor.classList.add("flor-rotando");
      flor.innerText = tiposFlores[i % tiposFlores.length];
      flor.style.top = Math.random() * 100 + "vh";
      flor.style.left = Math.random() * 100 + "vw";
      flor.style.fontSize = (Math.floor(Math.random() * 25) + 20) + "px";
      flor.style.animationDuration = (Math.floor(Math.random() * 12) + 8) + "s";
      contenedorFlores.appendChild(flor);
    }

    // Animación de escritura y tarjetas al entrar al sitio
    const checkboxBienvenida = document.getElementById("cerrar-bienvenida");
    const textoTitulo = "Mi Sitio de Informática";
    const elementoTitulo = document.getElementById("titulo-animado");
    let animacionIniciada = false;

    function iniciarAnimaciones() {
      irAlInicio();
      if (animacionIniciada) return;
      animacionIniciada = true;

      let indice = 0;
      function escribirTitulo() {
        if (indice < textoTitulo.length) {
          elementoTitulo.innerHTML += textoTitulo.charAt(indice);
          indice++;
          setTimeout(escribirTitulo, 100);
        }
      }
      escribirTitulo();

      const tarjetas = document.querySelectorAll('.tarjeta');
      tarjetas.forEach((tarjeta, index) => {
        setTimeout(() => {
          tarjeta.classList.add('animada');
        }, 250 * (index + 1));
      });
    }

    checkboxBienvenida.addEventListener("change", function() {
      if (checkboxBienvenida.checked) {
        iniciarAnimaciones();
      } else {
        reiniciarCuestionario();
      }
    });

    // Menú desplegable al hacer clic
    const botonMenu = document.querySelector(".dropbtn");
    const listaMenu = document.querySelector(".contenido-dropdown");

    botonMenu.addEventListener("click", function(evento) {
      evento.stopPropagation();
      listaMenu.classList.toggle("mostrar");
    });

    window.addEventListener("click", function() {
      if (listaMenu.classList.contains("mostrar")) {
        listaMenu.classList.remove("mostrar");
      }
    });

    // Evaluación de la encuesta
    function evaluarEncuesta() {
      const totalPreguntas = 10;
      let puntajeTotal = 0;

      for (let i = 1; i <= totalPreguntas; i++) {
        document.getElementById(`bloque-p${i}`).classList.remove("pregunta-incompleta");
      }

      for (let i = 1; i <= totalPreguntas; i++) {
        const respuesta = document.querySelector(`input[name="p${i}"]:checked`);
        if (!respuesta) {
          const bloqueFaltante = document.getElementById(`bloque-p${i}`);
          bloqueFaltante.classList.add("pregunta-incompleta");
          bloqueFaltante.scrollIntoView({ behavior: "smooth", block: "center" });
          alert(`⚠️ ¡Atención! Es obligatorio contestar todas las preguntas.\n\nTe falta responder la pregunta #${i}.`);
          return;
        }
        puntajeTotal += parseInt(respuesta.value, 10);
      }

      const modal = document.getElementById("modal-resultado");
      const textoModal = document.getElementById("texto-resultado-modal");
      textoModal.className = "caja-resultado-modal";

      if (puntajeTotal >= 24) {
        textoModal.classList.add("nivel-bueno");
        textoModal.innerHTML = `<strong>🎉 ¡Excelente Salud Digital! (${puntajeTotal}/30 puntos)</strong><br><br>Mantienes un equilibrio muy sano entre el uso de la tecnología, tu descanso, tu postura y tu convivencia diaria. ¡Sigue así!`;
      } else if (puntajeTotal >= 17) {
        textoModal.classList.add("nivel-medio");
        textoModal.innerHTML = `<strong>⚠️ Nivel Moderado (${puntajeTotal}/30 puntos)</strong><br><br>Vas por buen camino, pero algunos hábitos pueden mejorar. Procura hacer pausas visuales más seguidas, cuidar tu postura y desconectarte antes de dormir.`;
      } else {
        textoModal.classList.add("nivel-malo");
        textoModal.innerHTML = `<strong>🚨 ¡Advertencia de Riesgo Digital! (${puntajeTotal}/30 puntos)</strong><br><br>El uso excesivo de pantallas está afectando tu descanso, concentración o postura. Te recomendamos establecer horarios sin teléfono y realizar actividades fuera de línea.`;
      }

      modal.style.display = "flex";
    }

    function cerrarModalResultado() {
      document.getElementById("modal-resultado").style.display = "none";
    }
  </script>
